cmake_minimum_required(VERSION 3.28)
project(ChiakiVisionCore LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_DNN OFF CACHE BOOL "" FORCE)
set(OPUS_DRED OFF CACHE BOOL "" FORCE)
set(OPUS_OSCE OFF CACHE BOOL "" FORCE)
add_subdirectory(Dependencies/mbedtls EXCLUDE_FROM_ALL)
add_subdirectory(Dependencies/opus EXCLUDE_FROM_ALL)

get_filename_component(CHIAKI_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(THIRD "${CHIAKI_ROOT}/third-party")
add_library(nanopb STATIC ${THIRD}/nanopb/pb_common.c ${THIRD}/nanopb/pb_decode.c ${THIRD}/nanopb/pb_encode.c)
target_include_directories(nanopb PUBLIC ${THIRD}/nanopb)
target_compile_definitions(nanopb PUBLIC PB_C99_STATIC_ASSERT)
set(GF_SOURCES gf gf_wgen gf_w4 gf_w8 gf_w16 gf_w32 gf_w64 gf_w128 gf_rand gf_general gf_cpu)
list(TRANSFORM GF_SOURCES PREPEND "${THIRD}/gf-complete/src/")
list(TRANSFORM GF_SOURCES APPEND ".c")
add_library(gf_complete STATIC ${GF_SOURCES})
target_include_directories(gf_complete PUBLIC ${THIRD}/gf-complete/include)
set(JERASURE_SOURCES galois jerasure reed_sol cauchy liberation)
list(TRANSFORM JERASURE_SOURCES PREPEND "${THIRD}/jerasure/src/")
list(TRANSFORM JERASURE_SOURCES APPEND ".c")
add_library(jerasure STATIC ${JERASURE_SOURCES})
target_include_directories(jerasure PUBLIC ${THIRD}/jerasure/include)
target_link_libraries(jerasure PUBLIC gf_complete)

find_program(PROTOC protoc REQUIRED NO_CMAKE_FIND_ROOT_PATH)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(PROTO_OUT "${CMAKE_CURRENT_BINARY_DIR}/protobuf")
file(MAKE_DIRECTORY ${PROTO_OUT})
add_custom_command(OUTPUT ${PROTO_OUT}/takion.pb.c ${PROTO_OUT}/takion.pb.h
 COMMAND ${PROTOC} -I${CHIAKI_ROOT}/lib/protobuf --descriptor_set_out=${PROTO_OUT}/takion.pb ${CHIAKI_ROOT}/lib/protobuf/takion.proto
 COMMAND ${Python3_EXECUTABLE} ${THIRD}/nanopb/generator/nanopb_generator.py -D${PROTO_OUT} ${PROTO_OUT}/takion.pb
 DEPENDS ${CHIAKI_ROOT}/lib/protobuf/takion.proto VERBATIM)
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/include/chiaki)
file(COPY ${CMAKE_CURRENT_LIST_DIR}/Dependencies/opus/include/ DESTINATION ${CMAKE_CURRENT_BINARY_DIR}/include/opus FILES_MATCHING PATTERN "*.h")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/include/chiaki/config.h "#pragma once\n#define CHIAKI_LIB_ENABLE_OPUS 1\n#define CHIAKI_LIB_ENABLE_PI_DECODER 0\n#define CHIAKI_LIB_ENABLE_FFMPEG_DECODER 0\n")
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS ${CHIAKI_ROOT}/lib/src/*.c)
list(FILTER CORE_SOURCES EXCLUDE REGEX "/(aia|aia_openssl|ffmpegdecoder|pidecoder)\\.c$")
add_library(chiaki-lib STATIC ${CORE_SOURCES}
 ${CHIAKI_ROOT}/lib/src/remote/rudp.c ${CHIAKI_ROOT}/lib/src/remote/rudpsendbuffer.c
 Native/LocalNetworkOnly.c Native/RemotePlayBridge.c ${PROTO_OUT}/takion.pb.c)
target_compile_definitions(chiaki-lib PUBLIC CHIAKI_LIB_ENABLE_MBEDTLS MBEDTLS_ALLOW_PRIVATE_ACCESS CHIAKI_LOCAL_ONLY)
target_include_directories(chiaki-lib PUBLIC ${CHIAKI_ROOT}/lib/include ${CMAKE_CURRENT_BINARY_DIR}/include PRIVATE ${PROTO_OUT})
target_link_libraries(chiaki-lib PUBLIC mbedcrypto opus nanopb jerasure)
if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin")
 target_link_libraries(chiaki-lib PUBLIC "-framework CoreServices")
endif()
set_target_properties(chiaki-lib PROPERTIES OUTPUT_NAME chiaki)

if(NOT CMAKE_CROSSCOMPILING)
 enable_testing()
 set(CHIAKI_LIB_ENABLE_MBEDTLS ON)
 set(CHIAKI_ENABLE_FFMPEG_DECODER OFF)
 add_subdirectory(${CHIAKI_ROOT}/test upstream-tests)
 add_executable(vision-core-test Tests/CoreTests.c)
 target_include_directories(vision-core-test PRIVATE Native)
 target_link_libraries(vision-core-test PRIVATE chiaki-lib)
 add_test(NAME vision-core COMMAND vision-core-test)
endif()
